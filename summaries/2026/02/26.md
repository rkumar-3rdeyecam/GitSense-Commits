import { Context } from "aws-lambda";
import { Logger } from "@aws-lambda-powertools/logger";
import { DriverNameFetcherEvent, DriverNameFetcherResponse } from "@common/CoachingDownloadInterfaces";
import { fetchDriverDetails } from "./CoachingDownloadBackend";

// Initialize logger
const logger = new Logger({ serviceName: "DriverNameFetcherLambda" });

/**
 * AWS Lambda handler for fetching driver names from private API
 */
export const handler = async (
  event: DriverNameFetcherEvent,
  context: Context
): Promise<DriverNameFetcherResponse> => {
  logger.addContext(context);
  try {
    logger.info("DriverNameFetcher Lambda invoked", { event });
    const { batchId, workflowId, uniqueDriverIds, tenantId, coachingIdToDriverId } = event;

    if (!batchId || !workflowId || !tenantId) {
      throw new Error("Missing required fields: batchId, workflowId, or tenantId");
    }

    // Handle empty uniqueDriverIds gracefully - return empty mapping if no driver IDs provided
    if (!uniqueDriverIds || uniqueDriverIds.length === 0) {
      logger.info("No driver IDs to fetch", { batchId, workflowId });
      const response: DriverNameFetcherResponse = {
        batchId,
        workflowId,
        coachingIdToDriverName: {}, // Return empty mapping
      };
      return response;
    }

    // Filter out empty/null values and deduplicate
    const validDriverIds = [...new Set(
      uniqueDriverIds.filter(id => id && id.trim())
    )];

    logger.info("Fetching driver details", {
      batchId,
      providedDriverIds: uniqueDriverIds.length,
      uniqueDriverIds: validDriverIds.length, // After deduplication
      tenantId
    });

    // Create mapping from coachingId to driver name
    const coachingIdToDriverName: Record<string, string> = {};

    // Process drivers in parallel with limited concurrency
    const batchSize = 10; // Process 10 requests at a time
    for (let i = 0; i < validDriverIds.length; i += batchSize) {
      const batchDriverIds = validDriverIds.slice(i, i + batchSize);
      const promises = batchDriverIds.map(async (driverId) => {
        try {
          const driverDetails = await fetchDriverDetails(driverId, tenantId);
          let driverName = driverId; // Fallback to driverId
          if (driverDetails && driverDetails.firstName && driverDetails.lastName) {
            driverName = `${driverDetails.firstName} ${driverDetails.lastName}`;
            logger.debug("Successfully fetched driver name", { driverId, driverName });
          } else if (driverDetails && driverDetails.name) {
            // Fallback to 'name' field if firstName/lastName not available
            driverName = driverDetails.name;
            logger.debug("Successfully fetched driver name from 'name' field", { driverId, driverName });
          } else {
            logger.warn("Failed to fetch driver name, using ID as fallback", { driverId });
          }

          // Map all coaching IDs that use this driver to the driver name
          Object.entries(coachingIdToDriverId).forEach(([coachingId, dId]) => {
            if (dId === driverId) {
              coachingIdToDriverName[coachingId] = driverName;
            }
          });
        } catch (error) {
          logger.error("Error fetching driver details", {
            driverId,
            error: error instanceof Error ? error.message : error
          });
          // Use driverId as fallback for all coaching IDs using this driver
          Object.entries(coachingIdToDriverId).forEach(([coachingId, dId]) => {
            if (dId === driverId) {
              coachingIdToDriverName[coachingId] = driverId;
            }
          });
        }
      });
      // Wait for current batch to complete before processing next batch
      await Promise.all(promises);
    }

    const response: DriverNameFetcherResponse = {
      batchId,
      workflowId,
      coachingIdToDriverName,
    };

    logger.info("Driver details fetch completed", {
      batchId,
      totalRequested: validDriverIds.length,
      fetchedCount: Object.keys(coachingIdToDriverName).length
    });
    return response;
  } catch (error) {
    logger.error("Error in driver details fetcher", {
      error: error instanceof Error ? error.message : error,
      batchId: event.batchId,
      workflowId: event.workflowId
    });
    throw error;
  }
};

/**
 * Function to create a DynamoDb table for Coaching
 *
 * @function createCoachingTable
 * @param {Stack} stack
 * @returns dynamodb.TableV2
 */
export const createCoachingTable = (stack: Stack): TableV2 => {
  const biStream = `arn:aws:kinesis:${stack.region}:${stack.account}:stream/ArkBiKinesisStream`;
  let importedBiKinesisStream: IStream | undefined;

  const hasBiStream = ssm.StringParameter.valueFromLookup(
    stack,
    "/HasBIStream"
  );

  if (hasBiStream === "true") {
    importedBiKinesisStream = Stream.fromStreamArn(
      stack,
      "ExistingBIStream",
      biStream
    );
  }

  // Datastore for Coaching
  const coachingTable = new TableV2(stack, COACHING_DB, {
    partitionKey: { name: COACHING_DB_COL_PK, type: AttributeType.STRING },
    sortKey: { name: COACHING_DB_COL_SK, type: AttributeType.STRING },
    billing: Billing.onDemand(),
    encryption: TableEncryptionV2.dynamoOwnedKey(),
    tableClass: TableClass.STANDARD,
    tableName: COACHING_DB,
    removalPolicy: RemovalPolicy.RETAIN_ON_UPDATE_OR_DELETE,
    dynamoStream: StreamViewType.NEW_AND_OLD_IMAGES,
    kinesisStream: hasBiStream === "true" ? importedBiKinesisStream : undefined,
    pointInTimeRecovery: true,
  });

  // Add a Global Secondary Index (GSI) on the 'driverId' column
  coachingTable.addGlobalSecondaryIndex({
    indexName: COACHING_DB_INDEX_DRIVER_ID,
    partitionKey: { name: DRIVER_ID, type: AttributeType.STRING },
    sortKey: { name: COACHING_DB_COL_PK, type: AttributeType.STRING },
    projectionType: ProjectionType.ALL,
  });

  // Add another GSI on the 'completionDate' column
  coachingTable.addGlobalSecondaryIndex({
    indexName: COACHING_DB_INDEX_COMPLETION_DATE,
    partitionKey: { name: COACHING_DB_COL_PK, type: AttributeType.STRING },
    sortKey: { name: COMPLETION_DATE, type: AttributeType.STRING },
    projectionType: ProjectionType.ALL,
  });

  // Add another GSI on the 'coachId' column
  coachingTable.addGlobalSecondaryIndex({
    indexName: COACHING_DB_INDEX_COACH_ID,
    partitionKey: { name: COACHING_DB_COL_PK, type: AttributeType.STRING },
    sortKey: { name: COACH_ID, type: AttributeType.STRING },
    projectionType: ProjectionType.ALL,
  });

  return coachingTable;
};

/**
 * Function to create an S3 bucket for coaching exports
 *
 * @function createCoachingExportBucket
 * @param {Stack} stack
 * @returns s3.Bucket
 */
export const createCoachingDownloadBucket = (stack: Stack): s3.Bucket => {
  const arkEnv = process.env.AWS_ENVIRONMENT || "dev";

  // Create S3 bucket
  const coachingDownloadBucket = new s3.Bucket(
    stack,
    S3_COACHING_DOWNLOAD_BUCKET,
    {
      bucketName: S3_COACHING_DOWNLOAD_BUCKET + "-" + arkEnv,
      objectOwnership: s3.ObjectOwnership.BUCKET_OWNER_ENFORCED,
      removalPolicy: cdk.RemovalPolicy.RETAIN,
      blockPublicAccess: s3.BlockPublicAccess.BLOCK_ALL,
      encryption: s3.BucketEncryption.S3_MANAGED,
      enforceSSL: true,
      lifecycleRules: [
        {
          id: "DeleteWorkflowFilesAfter7Days",
          enabled: true,
          expiration: cdk.Duration.days(7),
          prefix: "workflows/",
        },
        {
          id: "DeleteRequestsAfter7Days",
          enabled: true,
          expiration: cdk.Duration.days(7),
          prefix: "workflow-requests/",
        },
        {
          id: "DeleteDownloadsAfter1Day",
          enabled: true,
          expiration: cdk.Duration.days(1),
          prefix: "downloads/",
        }
      ],
      versioned: false,
    }
  );

  // ssl is enforced
  const downloadBucketSslPolicyStatement = new cdk.aws_iam.PolicyStatement({
    effect: Effect.DENY,
    actions: ["s3:*"],
    principals: [new cdk.aws_iam.AnyPrincipal()],
    resources: [coachingDownloadBucket.arnForObjects("*")],
    conditions: {
      Bool: {
        "aws:SecureTransport": "false",
      },
    },
  });

  coachingDownloadBucket.addToResourcePolicy(downloadBucketSslPolicyStatement);

  return coachingDownloadBucket;
};

export async function generateCsv(
  safetyEvents: SafetyEvent[],
  coachingRows: CoachingData[],
): Promise<string> {
  try {
    const coachingMap = new Map<string, CoachingData>();
    coachingRows.forEach(row => {
      coachingMap.set(row.coachingId, row);
    });

    const safetyEventMap = new Map<string, SafetyEvent>();
    safetyEvents.forEach(event => {
      if (event.coachingId) {
        safetyEventMap.set(event.coachingId, event);
      }
    });

    // Always map over coachingRows, merging with safetyEvents if present
    const combinedData: CombinedCoachingData[] = coachingRows.map(coachingData => {
      const safetyEvent = safetyEventMap.get(coachingData.coachingId);
      return {
        correlationId: safetyEvent?.correlationId ?? '',
        timeOfEvent: safetyEvent?.timeOfEvent ?? '',
        coachingId: coachingData.coachingId,
        infractionDetails: safetyEvent?.infractionDetails ?? {}, // Ensure infractionDetails is an object
        completionDate: coachingData.completionDate,
        driverId: coachingData.driverId || safetyEvent?.driverId, // Use safety event driverId as fallback
        coachId: coachingData.coachId,
        coachingNotes: coachingData.coachingNotes,
        coachName: coachingData.coachName as string | undefined,
        driverName: coachingData.driverName as string | undefined
      };
    });

    logger.debug("✨CoachingBackend 🌿generateCsv 🍀combinedData created", {
      totalCombinedRecords: combinedData.length,
      sampleCombined: combinedData.length > 0 ? combinedData[0] : null
    });

    // CSV header
    const csvLines = [
      'EventId,EventDate,Session ID,Performed,Event Details,Coach,Driver,Notes'
    ];

    // Process combined data to generate CSV rows
    combinedData.forEach(combined => {
      // Format infractions from safety event infractionDetails as reviewType only
      const infractions = Array.isArray(combined.infractionDetails?.infractions)
        ? combined.infractionDetails.infractions
            .map(infraction => `${infraction.reviewType}`)
            .join(', ')
        : '';

      // Escape CSV values (handle commas and quotes)
      const escapeCSV = (value: unknown): string => {
        const str = String(value || '');
        if (str.includes(',') || str.includes('"') || str.includes('\n')) {
          return `"${str.replace(/"/g, '""')}"`;
        }
        return str;
      };

      const csvRow = [
        escapeCSV(combined.correlationId),
        escapeCSV(combined.timeOfEvent),
        escapeCSV(combined.coachingId),
        escapeCSV(combined.completionDate),
        escapeCSV(infractions),
        escapeCSV(combined.coachName ?? combined.coachId), // Use coachName if present, otherwise coachId
        escapeCSV(combined.driverName ?? combined.driverId), // Use driverName if present, otherwise driverId
        escapeCSV(combined.coachingNotes)
      ].join(',');
      csvLines.push(csvRow);
    });

    // Generate CSV content in memory
    const csvContent = csvLines.join('\n');

    logger.debug("✨CoachingBackend 🌿generateCsv ✅completed", {
      totalRows: csvLines.length - 1, // excluding header
      combinedRecordsCount: combinedData.length,
      originalSafetyEventsCount: safetyEvents.length,
      originalCoachingDataCount: coachingRows.length,
      csvSizeBytes: csvContent.length
    });

    return csvContent;
  } catch (error) {
    logger.error("✨CoachingBackend 🌿generateCsv ⚠️error", { error });
    throw error;
  }
}

    const body = event.body ?? "{}";
    const downloadrequest: DownloadCoachingRequest = JSON.parse(body);
    const tenantId = event.queryStringParameters?.tenantId;

    logger.debug("DownloadCoachingRequestLambdaFn 🌿handleDownloadCoachingSession 🔰request", { downloadrequest, tenantId });

    if (!tenantId) {
      return badRequest("tenantId is required");
    }
    if (!downloadrequest.userId) {
      return badRequest("userId is required");
    }
    if (!downloadrequest.CoachingIds || !Array.isArray(downloadrequest.CoachingIds) || downloadrequest.CoachingIds.length === 0) {
      return badRequest("CoachingIds[] is required and cannot be empty");
    }

    // Deduplicate CoachingIds
    const uniqueCoachingIds = Array.from(new Set(downloadrequest.CoachingIds));
    logger.debug("Processing download via workflow", { keyCount: uniqueCoachingIds.length, originalKeyCount: downloadrequest.CoachingIds.length });

    // Use deduplicated CoachingIds for workflow
    const dedupedRequest = { ...downloadrequest, CoachingIds: uniqueCoachingIds };

    const workflowResult = await initiateWorkflowDownload(dedupedRequest, tenantId);
    return ok(workflowResult);
  } catch (error) {
    logger.error("✨DownloadCoachingRequestLambdaFn 🌿handleDownloadCoachingSession ⚠️error", { error });
    return failedWithError(error);
  }

/**
 * state machine workflow-based coaching session download
 */
async function initiateWorkflowDownload(request: DownloadCoachingRequest, tenantId: string): Promise<{ downloadWorkflowId: string; message: string; }> {
  const workflowId = `coaching-download-${uuidv4()}`;
  try {
    // Store request details in S3 for workflow processing
    const coachingIdsS3Key = await storeCoachingIdsToS3(workflowId, request.CoachingIds, tenantId, request.userId);

    // Get Step Functions State Machine ARN from environment
    const stateMachineArn = process.env.COACHING_DOWNLOAD_STATE_MACHINE_ARN;
    if (!stateMachineArn) {
      throw new Error("COACHING_DOWNLOAD_STATE_MACHINE_ARN environment variable not set");
    }

    const userId = request.userId;

    // Input for Step Functions workflow
    const stepFunctionInput = {
      workflowId,
      coachingIdsS3Key, // S3 location of coaching IDs
      tenantId,
      userId,
    };

    // Step Functions execution
    const startResult = await sfnClient.send(new StartExecutionCommand({
      stateMachineArn,
      name: workflowId,
      input: JSON.stringify(stepFunctionInput),
    }));

    logger.info("Workflow download initiated successfully", {
      workflowId,
      executionArn: startResult.executionArn,
      userId,
      tenantId,
      keyCount: request.CoachingIds.length,
    });

    return {
      downloadWorkflowId : workflowId,
      message: `Download workflow started for ${request.CoachingIds.length} coaching sessions. You will receive a notification when the download is ready.`
    };

  } catch (error) {
    logger.error("✨DownloadCoachingRequestLambdaFn 🌿initiateWorkflowDownload ⚠️error", {
      error: error instanceof Error ? error.message : error,
      workflowId,
      keyCount: request.CoachingIds.length,
    });

    // Sending failure notification to EventBridge for notification service to pick up and notify user
    await sendWorkflowFailureNotification(tenantId, request.userId, error instanceof Error ? error.message : "Unknown error")
      .catch(notifError => logger.error("✨DownloadCoachingRequestLambdaFn 🌿sendWorkflowFailureNotification ⚠️error", { notifError }));
    throw error;
  }
}

  _context: Context

  let tenantId: string = '';
  let userId: string = '';

  try {
    // Extract workflowId, tenantId, userId from the first available batch result
    // The event from a Distributed Map's 'ResultPath' with multiple iterations is an array of arrays.
    // E.g., [[{batch1_output}], [{batch2_output}], ...] or if a "Map" is used within a "Map"
    // We assume at least one batch will have a consistent workflowId, tenantId, userId.
    if (Array.isArray(event) && event.length > 0 && Array.isArray(event[0]) && event[0].length > 0 && event[0][0]) {
      const firstBatch = event[0][0];
      workflowId = firstBatch.workflowId;
      tenantId = firstBatch.tenantId;
      userId = firstBatch.userId;
    } else {
      logger.warn("Event format not as expected for Distributed Map output, attempting fallback to flat event structure.", { event });
      // Fallback for cases where event might be a single object directly (less common for distributed map result)
      const flatEvent = event as unknown as ConsolidatorCoachingCsvEvent; // Type assertion for fallback
      workflowId = flatEvent.workflowId || 'unknown';
      tenantId = flatEvent.tenantId || '';
      userId = flatEvent.userId || '';
    }

    logger.info("ConsolidatorCoachingCsv Lambda invoked", { workflowId, tenantId, userId, originalEvent: event });

    // Validate required fields
    if (!workflowId || !tenantId || !userId) {
      throw new Error("Missing required fields: workflowId, tenantId, or userId extracted from event");

      totalBatchesFound: batchFiles.length,
      workflowId,
      batchesPrefix,
      sampleFiles: batchFiles.slice(0, 3)
    });

    const consolidationResult = await consolidateCoachingCsvFilesFromS3(workflowId, batchFiles);

    const finalCsvS3Key = `downloads/${workflowId}/coaching-export-${timestamp}.csv`;

/**
 * Consolidate coaching CSV files from S3 folder
 * Skips the header for all files except the first one.
 */
async function consolidateCoachingCsvFilesFromS3(workflowId: string, csvS3Keys: string[]): Promise<{
  consolidatedCsv: string;
  totalRows: number;
  fileSizeBytes: number;
  successfulBatches: number;
  failedBatches: number;
}> {
  let consolidatedCsv = '';
  let totalRows = 0;
  let successfulBatches = 0;
  let failedBatches = 0;
  let headerWritten = false;

  try {
    for (const csvS3Key of csvS3Keys) {
      try {
        const csvContent = await getCoachingCsvFromS3(csvS3Key);
        const lines = csvContent.split('\n').filter(line => line.trim());

        if (lines.length > 0) {
          if (!headerWritten) {
            // Include header from first batch
            consolidatedCsv += lines[0] + '\n';
            headerWritten = true;
          }
          // Add data rows (skip header)
          const dataRows = lines.slice(1);
          if (dataRows.length > 0) {
            consolidatedCsv += dataRows.join('\n') + '\n';
            totalRows += dataRows.length;
          }
        }
        successfulBatches++;
      } catch (batchError) {
        logger.error("Error processing coaching batch CSV", {
          csvS3Key,
          error: batchError instanceof Error ? batchError.message : batchError,
        });
        failedBatches++;
      }
    }

    const fileSizeBytes = Buffer.byteLength(consolidatedCsv, 'utf8');
    logger.info("Coaching CSV consolidation completed", {
      workflowId,
      totalRows,
      fileSizeBytes,
      successfulBatches,
      failedBatches,
    });

    return {
      consolidatedCsv,
      totalRows,
      fileSizeBytes,
      successfulBatches,
      failedBatches,
    };
  } catch (error) {
    logger.error("Error in coaching CSV consolidation", {
      error: error instanceof Error ? error.message : error,
      workflowId,
      processedBatches: successfulBatches,
    });
    throw error;
  }
}
